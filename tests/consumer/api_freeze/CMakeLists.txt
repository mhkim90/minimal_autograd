cmake_minimum_required(VERSION 3.14)
project(autograd_api_freeze_consumer LANGUAGES CXX)

# Each version-selection control uses a fresh standalone configure. Rejections
# come from the real installed ConfigVersion file, before any compile probes.
set(_version_args "")
if(AUTOGRAD_REQUEST_VERSION)
    list(APPEND _version_args "${AUTOGRAD_REQUEST_VERSION}")
    if(AUTOGRAD_REQUEST_EXACT)
        list(APPEND _version_args EXACT)
    endif()
endif()
option(AUTOGRAD_FREEZE_EXPERT "Also prove removed names fail with explicit Eigen expert" OFF)
if(AUTOGRAD_FREEZE_EXPERT)
    find_package(autograd ${_version_args} CONFIG REQUIRED COMPONENTS expert)
    set(_probe_target autograd::autograd_expert)
else()
    find_package(autograd ${_version_args} CONFIG REQUIRED)
    set(_probe_target autograd::autograd)
    if(TARGET autograd::autograd_expert)
        message(FATAL_ERROR "component-free package exposes expert target")
    endif()
endif()
get_target_property(_capability autograd::autograd AUTOGRAD_CUDA_ENABLED)
if(NOT "${_capability}" STREQUAL "0" AND NOT "${_capability}" STREQUAL "1")
    message(FATAL_ERROR "AUTOGRAD_CUDA_ENABLED must be literal 0/1")
endif()
if(DEFINED AUTOGRAD_EXPECT_CUDA_ENABLED AND
   NOT "${_capability}" STREQUAL "${AUTOGRAD_EXPECT_CUDA_ENABLED}")
    message(FATAL_ERROR "AUTOGRAD_CUDA_ENABLED does not match expected value")
endif()
foreach(_target autograd::autograd ${_probe_target})
    get_target_property(_definitions ${_target} INTERFACE_COMPILE_DEFINITIONS)
    if(_definitions MATCHES "AUTOGRAD_(USE_CUDA|TEST_CUDA_ENABLED)")
        message(FATAL_ERROR "${_target} leaks a private backend/test macro")
    endif()
endforeach()

function(probe name source expect_success diagnostic)
    set(_file "${CMAKE_CURRENT_BINARY_DIR}/probes/${name}.cpp")
    file(WRITE "${_file}" "${source}\n")
    try_compile(_ok "${CMAKE_CURRENT_BINARY_DIR}/probes/${name}"
        "${_file}" LINK_LIBRARIES ${_probe_target}
        CXX_STANDARD 17 CXX_STANDARD_REQUIRED ON OUTPUT_VARIABLE _output)
    file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/probes/${name}.log" "${_output}")
    if(expect_success)
        if(NOT _ok)
            message(FATAL_ERROR "Positive compile ${name} failed:\n${_output}")
        endif()
    else()
        # Match the diagnostic category AND the exact quoted identifier, not
        # incidental single-letter fields in unrelated compiler errors.
        string(REPLACE "‘" "'" _diagnostic_output "${_output}")
        string(REPLACE "’" "'" _diagnostic_output "${_diagnostic_output}")
        if(name MATCHES "^removed_header_")
            string(REPLACE "." "[.]" _path "${diagnostic}")
            set(_reason "fatal error: (${_path}: No such file or directory|'${_path}' file not found)")
        else()
            set(_identifier "'${diagnostic}'")
            set(_reason "error: [^\n]*((has no member named|no member named) ${_identifier}|${_identifier} (is not a member of|does not name a type)|'([^'\n]*::)?${diagnostic}' is private within this context)")
        endif()
        if(_ok OR NOT _diagnostic_output MATCHES "${_reason}")
            message(FATAL_ERROR "Negative compile ${name} did not fail for ${diagnostic}:\n${_output}")
        endif()
    endif()
    message(STATUS "API freeze ${name}: expected result confirmed")
endfunction()

# Prove compiler, selected package, link and BOTH namespace spellings before
# expected-failure controls. No missing compiler/SDK can satisfy the negatives.
probe(positive "#include <autograd.h>\n#include <autograd/autograd.h>\n#include <type_traits>\nstatic_assert(std::is_same<ag::Linear,ag::nn::Linear>::value,\"alias\");\nint main(){ ag::Variable x(ag::Tensor::ones(ag::Shape{1}),true); ag::sum(x).backward(); return x.has_grad()?0:1; }" TRUE "")
if(AUTOGRAD_FREEZE_EXPERT)
    probe(eigen_positive "#include <autograd/extension/eigen.h>\nint main(){ Eigen::MatrixXf m(1,2); m.setOnes(); return ag::tensor_to_eigen(ag::tensor_from_eigen(m)).cols()==2?0:1; }" TRUE "")
endif()
set(_headers autograd.h autograd/autograd.h
    autograd/tensor.h autograd/shape.h autograd/device.h
    autograd/core/variable.h autograd/core/ops.h autograd/core/loss.h
    autograd/core/module.h autograd/core/optim.h autograd/core/diffusion.h
    autograd/core/complex.h autograd/core/fft.h autograd/core/fft_norm.h
    autograd/extension/custom_op.h)
foreach(_header IN LISTS _headers)
    string(REPLACE "/" "_" _name "${_header}")
    probe("header_${_name}" "#include <${_header}>\n#if defined(EIGEN_WORLD_VERSION) || defined(CUDART_VERSION) || defined(CUDA_VERSION) || defined(__CUDA_RUNTIME_H__) || defined(AUTOGRAD_USE_CUDA) || defined(AUTOGRAD_TEST_CUDA_ENABLED)\n#error normal_header_leak\n#endif\nint main(){return 0;}" TRUE "")
endforeach()
foreach(_name Mat Mats VarPtr Var Function ComplexVar ReLUModule SiLUModule SigmoidModule)
    set(_include "#include <autograd.h>\n")
    if(AUTOGRAD_FREEZE_EXPERT)
        string(APPEND _include "#include <autograd/extension/eigen.h>\n")
    endif()
    probe("removed_${_name}" "${_include}int main(){ ag::${_name}* removed = nullptr; (void)removed; }" FALSE "${_name}")
endforeach()
foreach(_header variable function ops module loss optim conv norm diffusion complex fft cuda_core)
    probe("removed_header_${_header}" "#include <autograd/${_header}.h>\nint main(){return 0;}" FALSE "autograd/${_header}.h")
endforeach()
foreach(_field data parents back_fn node_ cuda_data_ cuda_grad_)
    probe("private_${_field}" "#include <autograd.h>\nint main(){ag::Variable x; (void)x.${_field};}" FALSE "${_field}")
endforeach()
foreach(_pair "Linear:W" "Linear:b" "Sequential:layers" "Adam:m" "Adam:v")
    string(REPLACE ":" ";" _parts "${_pair}")
    list(GET _parts 0 _type)
    list(GET _parts 1 _field)
    probe("private_${_type}_${_field}" "#include <autograd.h>\nvoid check(ag::${_type}& x){(void)x.${_field};}\nint main(){return 0;}" FALSE "${_field}")
endforeach()

add_executable(autograd_api_freeze main.cpp)
target_link_libraries(autograd_api_freeze PRIVATE autograd::autograd)
enable_testing()
add_test(NAME autograd_installed_api_freeze COMMAND autograd_api_freeze)
